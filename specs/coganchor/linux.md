# `coganchor/linux`

Dependency-free bindings to the Linux facilities a supervised session is built on: syscall
numbers and the host register layout, the seccomp trap filter, ptrace, and a tracee's memory,
cwd and descriptors. It knows nothing of the anchor, the wire, or a target.

## API

```python
# linux/__init__.py -- the four modules below and nothing else

# syscalls.py -- one machine's syscall numbers and register layout
@dataclass(frozen=True, slots=True)
class Numbers:
    # This machine's number for each syscall a supervised session cares about: the ones that
    # exec, exit, open, stat, access, read a link, change or make or remove a directory,
    # unlink, rename, symlink, link, chmod, truncate, set times, connect, install a filter,
    # reach another process by pidfd, and read or write its memory. One int field apiece.
    def trapped(self) -> frozenset[int]: ...

@dataclass(frozen=True, slots=True)
class Arch:
    name: str
    audit_arch: int
    number_index: int
    number_regset: int | None
    result_index: int
    arg_indices: tuple[int, int, int, int, int, int]
    stack_index: int
    register_count: int
    red_zone: int
    entry_plants_enosys: bool
    o_directory: int
    numbers: Numbers

SUPPORTED_MACHINES: dict[str, Arch]
ARCH: Arch                          # this machine's, read at import
NR: Numbers                         # ARCH.numbers
TRAPPED_SYSCALLS: frozenset[int]    # what a supervised session asks seccomp to trap

def syscall_name(number: int) -> str: ...

# seccomp.py -- the trap filter
def build_program(numbers: Iterable[int]) -> bytes: ...
def install(numbers: Iterable[int]) -> None: ...

# ptrace.py -- the tracer's side
OPTIONS: int            # what setoptions asks for unless told otherwise
WALL: int
SYSCALL_STOP_SIG: int
SYSCALL_STOP_ENTRY: int
SYSCALL_STOP_EXIT: int
SYSCALL_STOP_SECCOMP: int
EVENT_FORK: int
EVENT_VFORK: int
EVENT_CLONE: int
EVENT_EXEC: int
EVENT_SECCOMP: int

class Registers:
    def __init__(self, buffer: ctypes.Array[ctypes.c_ulonglong]) -> None: ...
    @property
    def dirty(self) -> bool: ...
    @property
    def buffer(self) -> ctypes.Array[ctypes.c_ulonglong]: ...
    @property
    def vector(self) -> int: ...
    @property
    def renumbered(self) -> bool: ...
    @property
    def number_vector(self) -> int: ...
    @property
    def syscall_number(self) -> int: ...
    @syscall_number.setter
    def syscall_number(self, value: int) -> None: ...
    @property
    def result(self) -> int: ...
    @result.setter
    def result(self, value: int) -> None: ...
    @property
    def stack_pointer(self) -> int: ...
    def settled(self) -> None: ...
    def scratch(self, size: int) -> int: ...
    def arg(self, index: int) -> int: ...
    def signed_arg(self, index: int) -> int: ...
    def set_arg(self, index: int, value: int) -> None: ...

def traceme() -> None: ...
def blank() -> Registers: ...
def setoptions(pid: int, options: int = OPTIONS) -> None: ...
def getregs(pid: int, into: Registers | None = None) -> Registers: ...
def setregs(pid: int, registers: Registers) -> None: ...
def cont(pid: int, signal: int = 0) -> None: ...
def syscall(pid: int, signal: int = 0) -> None: ...
def syscall_stop_kind(pid: int) -> int | None: ...
def get_event_message(pid: int) -> int: ...

# procfs.py -- a tracee's memory, cwd and descriptors
PATH_MAX: int
MAX_ARG_STRLEN: int

class TraceeGoneError(OSError): ...
class UnresolvedPathError(OSError): ...

class Peek:
    def __init__(self, size: int = PATH_MAX) -> None: ...
    def cstring(self, pid: int, address: int) -> bytes | None: ...

def read_bytes(pid: int, address: int, size: int) -> bytes: ...
def write_bytes(pid: int, address: int, data: bytes) -> int: ...
def read_cstring(pid: int, address: int, limit: int = PATH_MAX) -> str | None: ...
def read_string_array(pid: int, address: int, limit: int = 65536) -> list[str]: ...
def working_directory(pid: int) -> str: ...
def fd_target(pid: int, fd: int) -> str: ...
def resolve_magic(pid: int, path: str) -> str: ...
def steal_fd(pid: int, fd: int) -> int: ...
```

## Requirements

- MUST import nothing but the standard library, and name nothing of `hmz` outside itself.
- MUST fail at import on a machine it has no register map for, naming the machines it has.
- MUST offer a syscall's number, arguments, result and stack architecture-neutrally.
- MUST trap only the numbers `install` was given, inherited by every descendant and thread.
- MUST tell a tracee that has gone from a path it could not resolve, raising for either.
- MUST NOT be reached on the road to a native-CLI session.
